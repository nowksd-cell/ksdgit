}

let h = 2166136261;

/* id 자동 부여 */
function assignIds(node, id){
node.id = id;
(node.children||[]).forEach((c,i)=>assignIds(c, id+'-'+i));
}
BRANCHES.forEach(b=>assignIds(b.node, b.id));
TRUNK.forEach((t,i)=>{ t.id = 't'+i; });

/** 기본 화면: 각 갈래 33세(루트)만 펼치고, 34세 이하는 접기 */
function applyDefaultCollapsed(){
BRANCHES.forEach(b => {
function walk(n, isRoot){
const kids = n.children || [];
// 루트(33세)는 펼침, 그 아래(34세~) 중 자식이 있는 노드는 접기
if(kids.length && !isRoot) collapsed[n.id] = true;
kids.forEach(c => walk(c, false));
}
walk(b.node, true);
});
}

function field(node, key){
const e = edits[node.id];
if(e && e[key] !== undefined) return e[key];
return node[key];
}
function isRemoved(id){ return !!removed[id]; }

function collectExtraIds(){
const ids = new Set();
function walk(n){
if(!n) return;
if(n.id) ids.add(n.id);
(n.children||[]).forEach(walk);
}
Object.keys(extraChildren).forEach(pid => (extraChildren[pid]||[]).forEach(walk));
return ids;
}
function isExtraNode(id){ return collectExtraIds().has(id); }

/** 착오로 추가한 자손을 extraChildren에서 완전히 제거 */
function removeExtraNode(id){
function pruneList(list){
if(!list) return false;
const idx = list.findIndex(n => n.id === id);
if(idx >= 0){ list.splice(idx, 1); return true; }
for(const n of list){
if(n.children && pruneList(n.children)) return true;
}
return false;
}
let found = false;
Object.keys(extraChildren).forEach(pid => {
if(pruneList(extraChildren[pid])){
found = true;
if(!(extraChildren[pid]||[]).length) delete extraChildren[pid];
}
});
Object.keys(extraChildren).forEach(k => {
if(k === id || k.startsWith(id + '-')) delete extraChildren[k];
});
delete edits[id];
delete removed[id];
dropFromSiblingOrder(id);
return found;
}

function dropFromSiblingOrder(id){
Object.keys(siblingOrder).forEach(pid => {
siblingOrder[pid] = (siblingOrder[pid]||[]).filter(x => x !== id);
if(!siblingOrder[pid].length) delete siblingOrder[pid];
});
delete siblingOrder[id];
}

/* ============ 저장소 (IndexedDB — 대용량 지원) ============ */
const DB_NAME = 'ilsun-kim-jokbo-db';
const DB_VERSION = 1;
const STORE_NAME = 'state';
const STATE_KEY = 'main';
const LEGACY_LS_KEY = 'ilsun-kim-jokbo-state-v2'; // 이전 localStorage 마이그레이션용

let dbPromise = null;

function openDB(){
if(dbPromise) return dbPromise;
dbPromise = new Promise((resolve, reject)=>{
try{
const req = indexedDB.open(DB_NAME, DB_VERSION);
req.onupgradeneeded = (e)=>{
const db = e.target.result;
if(!db.objectStoreNames.contains(STORE_NAME)){
db.createObjectStore(STORE_NAME);
}
};
req.onsuccess = ()=> resolve(req.result);
req.onerror = ()=> reject(req.error || new Error('IndexedDB open failed'));
req.onblocked = ()=> console.warn('IndexedDB open blocked');
}catch(err){
reject(err);
}
});
return dbPromise;
}

function idbGet(key){
return openDB().then(db=> new Promise((resolve, reject)=>{
try{
const tx = db.transaction(STORE_NAME, 'readonly');
const store = tx.objectStore(STORE_NAME);
const req = store.get(key);
req.onsuccess = ()=> resolve(req.result);
req.onerror = ()=> reject(req.error);
}catch(e){ reject(e); }
}));
}

function idbSet(key, value){
return openDB().then(db=> new Promise((resolve, reject)=>{
try{
const tx = db.transaction(STORE_NAME, 'readwrite');
const store = tx.objectStore(STORE_NAME);
const req = store.put(value, key);
req.onsuccess = ()=> resolve(true);
req.onerror = ()=> reject(req.error);
tx.oncomplete = ()=>{};
tx.onerror = ()=> reject(tx.error);
}catch(e){ reject(e); }
}));
}

return true;
}catch(e){
console.warn('localStorage migration skipped:', e);
return false;

}

function applyStateObject(parsed, options){
// options: { keepCollapsed: true } — 온라인 데이터 적용 시 기기별 collapsed 유지
if(!parsed || typeof parsed !== 'object') return;
edits = (parsed.edits && typeof parsed.edits === 'object') ? parsed.edits : {};
removed = (parsed.removed && typeof parsed.removed === 'object') ? parsed.removed : {};
extraChildren = (parsed.extraChildren && typeof parsed.extraChildren === 'object') ? parsed.extraChildren : {};
siblingOrder = (parsed.siblingOrder && typeof parsed.siblingOrder === 'object') ? parsed.siblingOrder : {};
if(!(options && options.keepCollapsed)){
collapsed = (parsed.collapsed && typeof parsed.collapsed === 'object') ? parsed.collapsed : {};
}
if(parsed.savedAt) lastSaveTime = parsed.savedAt;
if(typeof parsed.version === 'number') onlineVersion = parsed.version;
}

/** Promise에 타임아웃 적용 */
function withTimeout(promise, ms, label){
return Promise.race([
promise,
new Promise((_, reject) => setTimeout(() => reject(new Error((label||'요청') + ' 시간 초과 (' + ms + 'ms)')), ms))
]);
}

/** 온라인 DB에서 공유 상태 불러오기. 성공 시 상태 객체, 실패 시 null */
async function loadOnlineState(){
if(!sbClient) return null;
try{
const result = await withTimeout(
sbClient
.from(ONLINE_TABLE)
.select('id, data, version, updated_at')
.eq('id', ONLINE_ROW_ID)
.maybeSingle(),
8000,
'온라인 족보 불러오기'
);
const { data, error } = result;
if(error){
console.warn('online load error:', error.message || error);
return null;
}
if(!data || !data.data) return null;
const payload = (typeof data.data === 'string') ? JSON.parse(data.data) : data.data;
if(typeof data.version === 'number') payload.version = data.version;
else if(typeof payload.version !== 'number') payload.version = 1;
onlineVersion = payload.version;
onlineAvailable = true;
return payload;
}catch(e){
console.warn('loadOnlineState failed:', e);
return null;
}
}

/** 온라인 DB에 공유 상태 저장. 충돌 검사 포함.
* @returns {{ok:boolean, conflict?:boolean, msg?:string, version?:number}}
*/
async function saveOnlineState(sharedState){
if(!sbClient) return { ok: false, msg: 'Supabase 미설정' };
try{
// 1) 현재 서버 version 확인 (충돌 방지)
const { data: current, error: readErr } = await sbClient
.from(ONLINE_TABLE)
.select('version')
.eq('id', ONLINE_ROW_ID)
.maybeSingle();
if(readErr){
console.warn('online version check failed:', readErr);
// 테이블이 없거나 권한 문제일 수 있음 — 그래도 시도
}
const serverVer = (current && typeof current.version === 'number') ? current.version : 0;
if(serverVer > 0 && onlineVersion > 0 && serverVer > onlineVersion){
return {
ok: false,
conflict: true,
msg: '다른 기기에서 더 최신 족보가 저장되었습니다.\n최신 데이터를 다시 불러온 후 수정해 주세요.'
};
}
const newVersion = Math.max(serverVer, onlineVersion, 0) + 1;
const toSave = {
version: newVersion,
savedAt: sharedState.savedAt || new Date().toISOString(),
edits: sharedState.edits || {},
removed: sharedState.removed || {},
extraChildren: sharedState.extraChildren || {},
siblingOrder: sharedState.siblingOrder || {}
// collapsed는 온라인에 공유하지 않음
};
const row = {
id: ONLINE_ROW_ID,
data: toSave,
version: newVersion,
updated_at: new Date().toISOString()
};
const { error: upsertErr } = await sbClient
.from(ONLINE_TABLE)
.upsert(row, { onConflict: 'id' });
if(upsertErr){
console.error('online save error:', upsertErr);
return { ok: false, msg: upsertErr.message || '온라인 저장 실패' };
}
// 2) 저장 후 다시 읽어 확인
const verify = await loadOnlineState();
if(!verify || verify.version !== newVersion){
return { ok: false, msg: '저장 후 확인에 실패했습니다. 잠시 후 다시 시도해 주세요.' };
}
onlineVersion = newVersion;
onlineAvailable = true;
ignoreRealtimeUntil = Date.now() + 2500; // 본인 저장 이벤트로 인한 토스트 방지
return { ok: true, version: newVersion };
}catch(e){
console.error('saveOnlineState exception:', e);
return { ok: false, msg: e.message || '온라인 저장 중 오류' };
}
}

try{

// 관리자이고 Supabase가 준비된 경우에만 온라인 저장 시도

// 비관리자 또는 온라인 미설정: 로컬만

return localOk;
}

}
}

function esc(s){ return (s||'').toString().replace(/[&<>"]/g, c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c])); }

function cleanName(name){
return (name || '').replace(/^[(\[]?諱[)\]]?\s*/, '').trim();
}
function isLostRecord(hanja, hangul){
const h1 = (hanja || '').trim();
const h2 = (hangul || '').trim();
if(!h1 && !h2) return true;
if(h1.includes('失傳') || h2.includes('실전')) return true;
return false;
}

function matchesSearch(node){
if(!searchTerm) return false;
const isDec = !!field(node, 'deceased');
const hanja = field(node,'hanja'), hangul = field(node,'hangul');
const hasHwi = (node.gen <= 33 && !isLostRecord(hanja, hangul)) || !!field(node, 'hwi');
const extra = (isDec ? '사망 ' : '') + (hasHwi ? '諱 휘 ' : '');
const hay = [extra, field(node,'hangul'), field(node,'hanja'), field(node,'spouse'), field(node,'note')].join(' ').toLowerCase();
return hay.includes(searchTerm);
}
function subtreeHasMatch(node){
if(matchesSearch(node)) return true;
return (allChildren(node)).some(subtreeHasMatch);
}
function allChildren(node){
const kids = (node.children||[]).slice();
const extra = extraChildren[node.id] || [];
let list = kids.concat(extra).filter(c=>!isRemoved(c.id));
const custom = siblingOrder[node.id];
if(custom && custom.length){
const pos = new Map(custom.map((cid,i)=>[cid,i]));
list.sort((a,b)=>{
const ia = pos.has(a.id) ? pos.get(a.id) : 10000;
const ib = pos.has(b.id) ? pos.get(b.id) : 10000;
if(ia !== ib) return ia - ib;
return 0;
});
return list;
}
return sortByBirth(list);
}

/** 생년월일 오름차순. 미입력은 맨 뒤, 같은 값은 원래 순서 유지 */
function parseBirthKey(node){
const raw = String(field(node,'birth')||'').replace(/\D/g,'');
if(!raw) return null;
let y=0, m=0, d=0;
if(raw.length <= 2){
y = parseInt(raw, 10);
} else {
y = parseInt(raw.slice(0,2), 10);
if(raw.length >= 4) m = parseInt(raw.slice(2,4), 10) || 0;
if(raw.length >= 6) d = parseInt(raw.slice(4,6), 10) || 0;
}
return y * 10000 + m * 100 + d;
}
function sortByBirth(list){
const years = list.map(parseBirthKey).filter(k => k !== null).map(k => Math.floor(k/10000));
const wrap = years.some(y=>y>=70) && years.some(y=>y<=30);
return list.map((n,i)=>({n,i,k:parseBirthKey(n)})).sort((a,b)=>{
if(a.k === null && b.k === null) return a.i - b.i;
if(a.k === null) return 1;
if(b.k === null) return -1;
let ka = a.k, kb = b.k;
if(wrap){
const ya = Math.floor(ka/10000), yb = Math.floor(kb/10000);
if(ya < 50) ka += 1000000;
if(yb < 50) kb += 1000000;
}
if(ka !== kb) return ka - kb;
return a.i - b.i;
}).map(x=>x.n);
}

function findParentNode(childId){
let found = null;
function walk(n){
if(found) return;
const kids = (n.children||[]).concat(extraChildren[n.id]||[]);
if(kids.some(k=>k.id===childId)){ found = n; return; }
kids.forEach(walk);
}
BRANCHES.forEach(b=>walk(b.node));
return found;
}
